add_executable(sim_gimbal_probe probe.cpp)
set_target_properties(sim_gimbal_probe PROPERTIES AUTOMOC ON)
target_link_libraries(sim_gimbal_probe PRIVATE Lidar::SimGimbal)
target_compile_definitions(sim_gimbal_probe PRIVATE QT_NO_KEYWORDS)
# The motion model also compiles and runs without any Qt target or include directory.
add_executable(gimbal_motion_probe motion.cpp)
target_include_directories(gimbal_motion_probe PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../include")
target_compile_features(gimbal_motion_probe PRIVATE cxx_std_20)
foreach(probe IN ITEMS sim_gimbal_probe gimbal_motion_probe)
    if(MSVC)
        target_compile_options(${probe} PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(${probe} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME ${probe} COMMAND ${probe})
    set_tests_properties(${probe} PROPERTIES TIMEOUT 30)
endforeach()
if(WIN32)
    set_tests_properties(sim_gimbal_probe PROPERTIES ENVIRONMENT_MODIFICATION
        "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
endif()
foreach(header IN ITEMS SimGimbalProtocol SimGimbalHandle GimbalMotion)
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${header}_alone.cpp" CONTENT "#include <${header}.h>\n")
    target_sources(sim_gimbal_probe PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${header}_alone.cpp")
endforeach()
